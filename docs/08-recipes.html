4 · A re-entrant full-screen transition

const CLOSE = 700, HOLD = 260, OPEN = 700;

export default function Transition({ run, onMidpoint, onComplete }) {
  const [phase, setPhase] = useState('closing');
  const busy = useRef(false);
  const timers = useRef([]);

  useEffect(() => {
    if (!run || busy.current) return undefined;   // ignore double-clicks
    busy.current = true;

    if (window.matchMedia?.('(prefers-reduced-motion: reduce)').matches) {
      onMidpoint?.(); onComplete?.(); busy.current = false; return undefined;
    }

    const at = (ms, fn) => timers.current.push(setTimeout(fn, ms));
    setPhase('closing');
    at(CLOSE, () => { setPhase('covered'); onMidpoint?.(); });  // swap state here
    at(CLOSE + HOLD, () => setPhase('opening'));
    at(CLOSE + HOLD + OPEN, () => { busy.current = false; onComplete?.(); });

    return () => { timers.current.forEach(clearTimeout); timers.current = []; busy.current = false; };
  }, [run]);

  if (!run) return null;
  return <div className={`iris iris-${phase}`} aria-hidden="true" />;
}
/* Keyframes, NOT a transition on a var: var-substituted values do not
   interpolate. Note the clip direction: this element is the cover, so a large
   circle hides the screen. */
.iris { position: fixed; inset: 0; z-index: 120; pointer-events: auto;
        clip-path: circle(0% at 50% 50%); will-change: clip-path; }
@keyframes irisClose { from { clip-path: circle(0% at 50% 50%); } to { clip-path: circle(150% at 50% 50%); } }
.iris-closing { animation: irisClose 700ms cubic-bezier(.32,.72,0,1) forwards; }
.iris-covered { clip-path: circle(150% at 50% 50%); }
Mount it once, above the thing that changes

If the transition lives inside the view it swaps, the swap unmounts it and it replays. Render it as a sibling of the view switch, not a child of a branch.

5 · Undo/redo in fifteen lines

// Requires: every edit is a pure function returning a NEW document.
function commit(state, nextDoc, selection = state.selection) {
  if (nextDoc === state.doc) return state;
  return { doc: nextDoc, past: [...state.past, state.doc].slice(-80), future: [], selection };
}

function reducer(state, action) {
  switch (action.type) {
    case 'edit': return commit(state, applyEdit(state.doc, action));
    case 'undo':
      if (!state.past.length) return state;
      return { doc: state.past.at(-1), past: state.past.slice(0, -1),
               future: [state.doc, ...state.future], selection: [] };
    case 'redo': {
      if (!state.future.length) return state;
      const [next, ...rest] = state.future;
      return { doc: next, past: [...state.past, state.doc], future: rest, selection: [] };
    }
    default: return state;
  }
}

Works whenever your state is JSON-serialisable and reasonably small. Cap the history length so memory stays bounded.

6 · A keyboard map with one source of truth

// lib/shortcuts.js: the table, used by BOTH the dispatcher and the help sheet.
export const SHORTCUT_GROUPS = [
  { group: 'Place', items: [
    { keys: ['A'], action: 'add_component', label: 'Add a component' },
    { keys: ['W'], action: 'tool_wire',     label: 'Draw wires' },
  ]},
];

export function resolveShortcut(event) {
  const typing = ['INPUT', 'TEXTAREA', 'SELECT'].includes(event.target.tagName)
    || event.target.isContentEditable;
  if (typing) return null;                 // never steal keys from a field

  const ctrl = event.ctrlKey || event.metaKey;   // metaKey = ⌘ on macOS
  if (ctrl && event.key.toLowerCase() === 'z') return event.shiftKey ? 'redo' : 'undo';

  switch (event.key.toLowerCase()) {
    case 'a': return 'add_component';
    case 'w': return 'tool_wire';
    default: return null;
  }
}

Then one listener dispatches on the action id, and the help panel renders from the same array. The documentation cannot drift from the behaviour.

7 · Seeded random generation

// mulberry32: small, fast, good enough for content generation (NOT for crypto).
export function makeRng(seed) {
  let a = (seed >>> 0) || 1;
  const next = () => {
    a += 0x6d2b79f5;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
  return {
    next,
    int: (min, max) => Math.floor(next() * (max - min + 1)) + min,
    pick: (list) => list[Math.floor(next() * list.length)],
  };
}

Store (templateId, seed) and you can regenerate any piece of content exactly: bug reports become reproducible, and your database stores two columns instead of a blob.

8 · Pan and zoom on SVG

const [view, setView] = useState({ x: 0, y: 0, zoom: 1 });

// Screen → world
const toWorld = (e) => {
  const r = svgRef.current.getBoundingClientRect();
  return { x: (e.clientX - r.left - view.x) / view.zoom, y: (e.clientY - r.top - view.y) / view.zoom };
};

// Zoom toward the pointer: keep the world point under the cursor fixed.
const onWheel = (e) => {
  e.preventDefault();
  const r = svgRef.current.getBoundingClientRect();
  const sx = e.clientX - r.left, sy = e.clientY - r.top;
  setView((v) => {
    const zoom = Math.min(6, Math.max(0.35, v.zoom * Math.exp(-e.deltaY * 0.0015)));
    const k = zoom / v.zoom;
    return { zoom, x: sx - (sx - v.x) * k, y: sy - (sy - v.y) * k };
  });
};

// Wheel must be a non-passive NATIVE listener to allow preventDefault.
useEffect(() => {
  const el = svgRef.current;
  el.addEventListener('wheel', onWheel, { passive: false });
  return () => el.removeEventListener('wheel', onWheel);
}, [onWheel]);

10 · Safe local persistence

const KEYS = { settings: 'app.settings.v1' };     // version your keys

function read(key, fallback) {
  try { const raw = localStorage.getItem(key); return raw ? JSON.parse(raw) : fallback; }
  catch { return fallback; }                    // private mode, quota, bad JSON
}
function write(key, value) {
  try { localStorage.setItem(key, JSON.stringify(value)); } catch {}
}

// Always merge with defaults, so a new setting appears for existing users.
export const store = {
  getSettings: () => ({ ...DEFAULT_SETTINGS, ...read(KEYS.settings, {}) }),
  setSettings: (s) => write(KEYS.settings, s),
};

11 · A safe Supabase table

create table public.notes (
  id         uuid primary key default gen_random_uuid(),
  user_id    uuid not null references auth.users (id) on delete cascade,
  body       text not null,
  created_at timestamptz not null default now()
);

create index on public.notes (user_id, created_at desc);
alter table public.notes enable row level security;

create policy "read own"   on public.notes for select using (auth.uid() = user_id);
create policy "insert own" on public.notes for insert with check (auth.uid() = user_id);
create policy "update own" on public.notes for update using (auth.uid() = user_id)
                                                  with check (auth.uid() = user_id);
create policy "delete own" on public.notes for delete using (auth.uid() = user_id);

Four policies, one per operation. Miss one and that operation is simply denied, which is the correct failure direction.

13 · Provisional state (claimed vs earned)

// Let the user assert something, act on it, and let reality settle it.
export function claim(state, ids) {
  const next = { ...state };
  for (const id of ids) {
    if (next[id]?.attempts > 0) continue;      // never overwrite real evidence
    next[id] = { value: THRESHOLD + 0.05, attempts: 0, self: true };
  }
  return next;
}

export function observe(state, ids, passed) {
  const next = { ...state };
  for (const id of ids) {
    const prev = next[id] ?? { value: 0, attempts: 0, self: false };
    const value = passed ? Math.min(1, prev.value + (1 - prev.value) * 0.34)
                        : prev.self ? 0.2                       // claim refuted
                        : Math.max(0, prev.value * 0.7 - 0.05);
    next[id] = { value, attempts: prev.attempts + 1, self: false };
  }
  return next;
}
Where else this applies

Any onboarding that asks "how experienced are you?", any recommender that needs a cold start, any form that pre-fills from a guess. Acting on a claim and correcting it from behaviour is almost always better UX than an up-front test, and it is a much smaller feature.

14 · Themed confetti (canvas particles)

function runConfetti(canvas) {
  const ctx = canvas.getContext('2d');
  const dpr = Math.min(window.devicePixelRatio || 1, 2);   // crisp, but capped
  const resize = () => {
    canvas.width = canvas.clientWidth * dpr;
    canvas.height = canvas.clientHeight * dpr;
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);            // draw in CSS pixels
  };
  resize();

  const particles = [];
  const spawn = (x, y, angle, spread, count, power) => { /* push particles */ };

  let raf = 0, stopped = false;
  const frame = () => {
    if (stopped) return;
    ctx.clearRect(0, 0, canvas.clientWidth, canvas.clientHeight);
    for (let i = particles.length - 1; i >= 0; i--) {
      const p = particles[i];
      p.vy += 0.24;              // gravity
      p.vx *= 0.988; p.vy *= 0.988; // drag
      p.x += p.vx; p.y += p.vy; p.rot += p.vrot;
      if (p.y > canvas.clientHeight + 60) { particles.splice(i, 1); continue; }
      ctx.save(); ctx.translate(p.x, p.y); ctx.rotate(p.rot);
      drawGlyph(ctx, p);        // ← draw YOUR subject, not paper squares
      ctx.restore();
    }
    if (!particles.length) { stopped = true; return; }   // stop on its own
    raf = requestAnimationFrame(frame);
  };
  raf = requestAnimationFrame(frame);

  return () => { stopped = true; cancelAnimationFrame(raf); };   // cleanup
}
The design move

This app's confetti are tiny resistors, capacitors, LEDs, chips and lightning bolts in the schematic palette. Generic confetti says "a library did this"; subject-matter confetti says "someone made this for you". Same code, one different drawGlyph.

15 · Invariant tests over a whole dataset

test('every symbol is well formed and on-grid', () => {
  for (const symbol of SYMBOLS) {
    for (const unit of symbol.units) {
      for (const pin of unit.pins) {
        // (-30 % 10) is -0 in JS, so compare the magnitude.
        assert.ok(Math.abs(pin.x % GRID) === 0,

        assert.ok(PIN_TYPES.has(pin.type),
          `${symbol.id} pin ${pin.num}: unknown type "${pin.type}"`);
      }
    }
  }
});

test('every id referenced by content exists in the library', () => {
  const problems = [];
  for (const template of TEMPLATES) { /* collect selectors, verify each */ }
  assert.deepEqual(problems, [], `mismatches:\n${problems.join('\n')}`);

These two tests found two real bugs on their first run. Any project with a content library: game items, form schemas, product catalogues: should have their equivalent. Assert properties of the whole set, and put the identifying detail in the message.

Anti-patterns to avoid

Do notBecauseInstead
Hoist all state to the rootEvery change re-renders everythingKeep fast-changing state local
Mutate state objectsReact cannot detect itReturn new objects
Use array index as keyBreaks on reorderA stable id from the data
Derive state in an effectExtra render, easy to desyncCompute during render
Skip effect cleanupLeaks listeners and timersAlways return a teardown
Transition a property through a CSS varIt will not interpolate@keyframes with literal values
Ship rules only on the clientUsers can edit the clientEnforce in the database (RLS) too
Animate width/topForces layout every frametransform, opacity, clip-path
outline: none with no replacementUnusable by keyboardA visible focus-visible style

Starting your own project tomorrow

npm create vite@latest my-app -- --template react
cd my-app && npm install
npm install -D tailwindcss postcss autoprefixer && npx tailwindcss init -p
git init && git add -A && git commit -m "Scaffold"

Then, in this order:

  1. Write the design tokens (chapter 06) before any component.
  2. Write the domain model as plain data and pure functions, no React.
  3. Write a test for the first invariant that matters.
  4. Build the UI on top, one component at a time.
  5. Add persistence last, behind one branch, with the local path working first.

That order is not arbitrary: it is the order that keeps you able to change your mind. And being able to change your mind cheaply is what separates a project you finish from one you abandon.
